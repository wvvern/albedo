# Dusk on 1.8.9 with the Sharp render on and off at a field of view of 30 and 70, where both draw from the baked image. Run it after scene-legacy.txt:
# ./tools/screenshots.ps1 -Targets forge-1.8.9 -Shots sky-sharp-modern.txt -OutDir build\sky-sharp (the script takes this file for forge)
wait 60
profile minimal
run time set 6000
wait 40
sky dusk
wait 160
# Yaw 90 looks west (the Dusk glow), pitch -60 looks up to the stars.
fov 30
run tp @p 0 3000 0 0 -60
wait 40
screenshot sharp-01-on-stars-fov30
run tp @p 0 3000 0 90 -5
wait 40
screenshot sharp-02-on-west-fov30
fov 70
run tp @p 0 3000 0 0 -60
wait 40
screenshot sharp-03-on-stars-fov70
skyrender fast
wait 160
fov 30
run tp @p 0 3000 0 0 -60
wait 40
screenshot sharp-04-off-stars-fov30
quit
