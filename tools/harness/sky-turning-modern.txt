# The turning sky: Nebula and Galaxy compared with the CPU bake, Galaxy turned with the world time and with the real time,
# sharp and fast, and the Motion section of the sky editor. Run it after scene-modern.txt:
# ./tools/screenshots.ps1 -Targets neoforge-1.21.11,fabric-1.21.1 -Shots sky-turning-modern.txt -OutDir build\sky-turning
wait 60
profile minimal
run time set 18000
wait 40
fov 70
run tp @p 0 3000 0 -45 -46
sky nebula
wait 160
skycompare
wait 40
sky galaxy
wait 160
screenshot turning-01-still
skycompare
wait 40
# With the world clock at midnight, the sky is as the design is.
skyturn 100 height 45
wait 20
screenshot turning-02-world-midnight
# At sunrise, the sky has turned about a quarter turn around the pole in the north.
run time set 0
wait 20
screenshot turning-03-world-sunrise
skyrender fast
wait 40
screenshot turning-04-world-sunrise-fast
skyrender sharp
wait 40
# With the real clock at 10 turns a day, 30 s is a quarter turn.
skyclock 0
skyturn 100 speed 10 height 45 real
wait 20
screenshot turning-05-real-0s
skyclock 30
wait 20
screenshot turning-06-real-30s
skycompare
wait 40
# Around a pole at the zenith, the sky turns as a whole to the west.
skyturn 100 speed 10 height 90 real
run tp @p ~ ~ ~ 0 -10
wait 20
screenshot turning-07-zenith-pole-30s
settings open
wait 20
ui click Sky
wait 60
ui click Motion
wait 20
ui pointer off
wait 10
screenshot turning-08-settings-motion
# A click on the Follow time label sets the switch, and the Turning slider at its left end stops the sky and hides the options.
ui click Follow time
wait 10
ui pointer off
wait 10
screenshot turning-09-settings-follow
ui slide 0 sky.turning
wait 10
ui pointer off
wait 10
screenshot turning-10-settings-still
settings close
wait 10
quit
