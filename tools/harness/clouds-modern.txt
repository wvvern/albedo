# Clouds in each state of the config, seen from below the cloud layer. The game starts with the default config (quake-white,
# clouds hidden). Each "clouds shown|hidden" line writes "! clouds" to the log when the next frame does not draw them as expected.
# Run it after scene-modern.txt in place of shots.txt:
# ./tools/screenshots.ps1 -Targets fabric-26.3 -Shots clouds-modern.txt -OutDir build\clouds
# Add -Config <a copy of config.json> to start with the profiles of a player.
wait 60
run setblock 0 169 4 barrier
run tp @p 0.5 170 4.5 180 -15
wait 100
screenshot c01-start
mod off
wait 100
screenshot c02-off
clouds shown
mod on
wait 100
screenshot c03-on
clouds hidden
profile pastel
wait 100
screenshot c04-pastel
clouds shown
profile minimal
wait 100
screenshot c05-minimal
clouds hidden
sky dusk
wait 160
screenshot c06-minimal-dusk
clouds hidden
sky off
profile pastel
reload
wait 300
screenshot c07-pastel-after-reload
clouds shown
profile quake-white
wait 100
screenshot c08-quake-white
clouds hidden
quit
