# The turning sky between two ticks: Galaxy at Speed 10 with the world clock, at three partial ticks between the same two
# ticks, one tick apart as it runs, and some ticks apart with the daylight cycle stopped. Run it after scene-modern.txt:
# ./tools/screenshots.ps1 -Targets fabric-26.3,fabric-1.21.1 -Shots sky-turning-smooth-modern.txt -OutDir build\sky-turning-smooth
wait 60
profile minimal
run time set 18000
wait 40
fov 70
run tp @p 0 3000 0 -45 -46
sky galaxy
skyturn 0
wait 160
skycompare
wait 40
skyclock 0
skyturn 100 speed 10 height 45
wait 20
# Between the same two ticks, the partial ticks 0, 0.5 and 1 turn the sky by 0, a half and one tick of the world clock.
skypartial 0
wait 2
screenshot smooth-01-part-0
skypartial 0.5
wait 2
screenshot smooth-02-part-50
skypartial 1
wait 2
screenshot smooth-03-part-100
skypartial run
wait 2
screenshot smooth-04-run
wait 1
screenshot smooth-05-run-next-tick
# With the daylight cycle stopped, the sky stays still in each frame; 1.21.1 has the old name of the game rule.
run gamerule doDaylightCycle false
run gamerule advance_time false
wait 60
screenshot smooth-06-stopped
wait 7
screenshot smooth-07-stopped-later
wait 13
screenshot smooth-08-stopped-last
run gamerule doDaylightCycle true
run gamerule advance_time true
skyclock run
skycompare
wait 40
quit
