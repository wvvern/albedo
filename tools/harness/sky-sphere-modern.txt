# Nebula and Galaxy high above the world, at and below the horizon, with "Full sphere" on (the preset) and off. Run it after scene-modern.txt:
# ./tools/screenshots.ps1 -Targets fabric-26.3 -Shots sky-sphere-modern.txt -OutDir build\sky-sphere
wait 60
profile minimal
run time set 6000
wait 40
sky nebula
skyturn 0
wait 160
run tp @p 0 3000 0 180 -5
wait 40
screenshot sphere-01-nebula-on-horizon
run tp @p 0 3000 0 180 40
wait 40
screenshot sphere-02-nebula-on-below
settings open
wait 20
ui click Sky
wait 60
ui pointer off
wait 10
screenshot sphere-03-settings-on
ui click Full sphere
wait 60
ui pointer off
wait 10
screenshot sphere-04-settings-off
ui click Done
wait 160
run tp @p 0 3000 0 180 -5
wait 40
screenshot sphere-05-nebula-off-horizon
run tp @p 0 3000 0 180 40
wait 40
screenshot sphere-06-nebula-off-below
sky galaxy
skyturn 0
wait 160
run tp @p 0 3000 0 -45 -5
wait 40
screenshot sphere-07-galaxy-on-horizon
run tp @p 0 3000 0 -45 40
wait 40
screenshot sphere-08-galaxy-on-below
settings open
wait 20
ui click Sky
wait 60
ui click Full sphere
wait 10
ui click Done
wait 160
run tp @p 0 3000 0 -45 -5
wait 40
screenshot sphere-09-galaxy-off-horizon
run tp @p 0 3000 0 -45 40
wait 40
screenshot sphere-10-galaxy-off-below
quit
