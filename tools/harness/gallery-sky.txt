# Store page images of the custom skies and the sky editor. Run it after scene-modern.txt:
# ./tools/screenshots.ps1 -Targets fabric-26.3 -Shots gallery-sky.txt -OutDir build\gallery-sky
wait 60
profile minimal
run time set 6000
# A lantern in place of the torch: torch smoke drifts through the sky shots.
run setblock 0 -59 -6 lantern
wait 40
# Nebula over the scene, looking up a little.
sky nebula
skyturn 0
wait 160
run tp @p 0.5 -60 6.5 180 -14
wait 40
screenshot s01-nebula-a
run tp @p 0.5 -60 6.5 180 -22
wait 40
screenshot s01-nebula-b
# Galaxy: high in the south-east, so the camera stands north-west of the scene with the mobs in front of it.
sky galaxy
skyturn 0
run kill @e[type=!player]
wait 5
run kill @e[type=item]
run summon pig -10.2 -60 -8.6 {NoAI:1b,Rotation:[135f,0f]}
run summon sheep -8.8 -60 -10 {NoAI:1b,Color:14,Rotation:[135f,0f]}
run summon villager -6 -60 -12.8 {NoAI:1b,Rotation:[135f,0f]}
run summon cow -4.6 -60 -14.2 {NoAI:1b,Rotation:[135f,0f]}
wait 160
run tp @p -14.5 -60 -18.5 -45 -20
wait 40
screenshot s02-galaxy-a
run tp @p -14.5 -60 -18.5 -45 -24
wait 40
screenshot s02-galaxy-b
run tp @p -14.5 -60 -18.5 -35 -24
wait 40
screenshot s02-galaxy-c
# Dusk: the sun sets in the west, so the camera stands east of the scene with the mobs facing it.
sky dusk
skyturn 0
run kill @e[type=!player]
wait 5
run kill @e[type=item]
run summon pig 1 -60 3 {NoAI:1b,Rotation:[-62f,0f]}
run summon sheep 1.8 -60 1.5 {NoAI:1b,Color:14,Rotation:[-62f,0f]}
run summon villager 2.6 -60 0 {NoAI:1b,Rotation:[-62f,0f]}
run summon cow 3.4 -60 -1.5 {NoAI:1b,Rotation:[-62f,0f]}
run time set 12000
wait 160
run tp @p 12.5 -60 3.5 110 -6
wait 40
screenshot s03-dusk-a
run tp @p 12.5 -60 3.5 100 -8
wait 40
screenshot s03-dusk-b
run tp @p 14.5 -60 1.5 100 -4
wait 40
screenshot s03-dusk-c
run time set 6000
wait 40
# Back to the scene as built, for the editor.
run kill @e[type=!player]
wait 5
run kill @e[type=item]
run summon pig -3 -60 -1 {NoAI:1b,Rotation:[0f,0f]}
run summon sheep -1 -60 -1 {NoAI:1b,Color:14,Rotation:[0f,0f]}
run summon villager 1 -60 -1 {NoAI:1b,Rotation:[0f,0f]}
run summon cow 3 -60 -1 {NoAI:1b,Rotation:[0f,0f]}
run tp @p 0.5 -60 4.5 180 -10
sky nebula
skyturn 0
wait 160
settings open
wait 20
ui click Sky
wait 60
ui pointer off
wait 10
screenshot s04-editor-nebula
ui scroll -1 Bloom 7
wait 20
ui pointer off
wait 10
screenshot s04-editor-nebula-scrolled
settings close
wait 10
sky galaxy
skyturn 0
wait 160
settings open
wait 20
ui click Sky
wait 60
ui click Galaxy 4
wait 20
ui click This layer
wait 60
ui pointer off
wait 10
screenshot s05-layer-galaxy-alone
ui scroll -3 Opacity
wait 30
ui pointer off
wait 10
screenshot s05-layer-galaxy-look
ui click Place
wait 30
ui pointer off
wait 10
screenshot s05-layer-galaxy-place
