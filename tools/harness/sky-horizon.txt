# Dusk, Nebula and Galaxy toward the horizon at a field of view of 70, with "Sharp render" on and off and "Full sphere" on and off. Run it after the scene:
# ./tools/screenshots.ps1 -Targets fabric-26.3,fabric-1.21.11,forge-1.8.9 -Shots sky-horizon.txt -OutDir build\sky-horizon
wait 60
profile minimal
run time set 6000
wait 40
fov 70
# Nebula from the ground, where the fog of the world meets the sky.
sky nebula
wait 160
run tp @p ~ ~ ~ 180 0
wait 40
screenshot horizon-00-nebula-ground
# Yaw 90 looks west (the Dusk glow), yaw 180 looks north (the Nebula), yaw -45 looks south-east (the galaxy).
sky dusk
skyextent horizon
wait 160
run tp @p 0 3000 0 90 0
wait 40
screenshot horizon-01-dusk-off-sharp
skycompare
wait 40
skyrender fast
wait 40
screenshot horizon-02-dusk-off-fast
skyrender sharp
skyextent sphere
wait 160
screenshot horizon-03-dusk-on-sharp
skycompare
wait 40
skyrender fast
wait 40
screenshot horizon-04-dusk-on-fast
sky nebula
skyextent sphere
wait 160
run tp @p 0 3000 0 180 0
wait 40
screenshot horizon-05-nebula-on-sharp
skycompare
wait 40
skyrender fast
wait 40
screenshot horizon-06-nebula-on-fast
skyrender sharp
skyextent horizon
wait 160
screenshot horizon-07-nebula-off-sharp
skycompare
wait 40
skyrender fast
wait 40
screenshot horizon-08-nebula-off-fast
sky galaxy
skyextent sphere
wait 160
run tp @p 0 3000 0 -45 0
wait 40
screenshot horizon-09-galaxy-on-sharp
skycompare
wait 40
skyrender fast
wait 40
screenshot horizon-10-galaxy-on-fast
skyrender sharp
skyextent horizon
wait 160
screenshot horizon-11-galaxy-off-sharp
skycompare
wait 40
skyrender fast
wait 40
screenshot horizon-12-galaxy-off-fast
quit
