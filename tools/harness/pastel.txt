# The built-in pastel profile with its soft Chroma colors, which drift slowly: two screenshots 10 s apart, then frames in a row.
# Run it after the scene, for example before chroma.txt:
# ./tools/screenshots.ps1 -Targets fabric-26.3,forge-1.8.9 -Extra pastel.txt -Shots chroma.txt -OutDir build\chroma
wait 60
profile pastel
wait 200
screenshot p01-pastel
# One turn takes about 96 s, so the hue of each block moves about 37 degrees in 10 s.
wait 200
screenshot p02-pastel-10s
frames 3 p03-row
wait 20
settings open
wait 20
ui click Effects
wait 10
ui pointer off
wait 10
screenshot p04-settings-effects
settings close
wait 10
