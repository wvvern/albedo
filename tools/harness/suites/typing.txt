# Types into each text field of the settings screen through the keyboard path of the game, as a real keyboard does.
# Each "ui expect" logs PASS or a line that starts with "!". A "ui" line waits for the next frame, but a key goes in at
# once, so a "wait" follows each "ui" line that comes before a key. Run it after the scene:
# python -m tools harness fabric-26.3 forge-1.8.9 --suite typing --out build/typing
# A legacy target cannot do "ui realclick" and runs "ui click" in its place, and it skips the lines that start with "modern:".
wait 60
profile quake-white
wait 100
settings open
wait 20
# Esc with no focused field closes the screen.
key ESCAPE
wait 10
expect closed
settings open
wait 20
# Texture search: letters, digits, Backspace, the arrow keys, Home, End and Delete.
ui click Textures
wait 10
ui realclick Search textures
wait 10
type oak2
wait 5
ui expect search oak2
wait 3
key BACKSPACE
key LEFT
key LEFT
type x
key HOME
type y
key END
type z
key HOME
key DELETE
key RIGHT
key BACKSPACE
wait 5
ui expect search xakz
ui pointer off
wait 5
screenshot t01-search
# Ctrl+A selects all and the next letter replaces it; E is the inventory key and F6, F7 and F9 are the keys of Albedo.

wait 2

wait 2

type e
modern: key F6
modern: key F7
key F9
wait 5

wait 3

# AltGr is Control and Alt on Windows, as for | on a German keyboard.

wait 2

keyup LEFT_CONTROL
wait 3
