# The Galaxy preset at a field of view of 70 and 30, at day and at night, from high up and from the ground, with the GPU sky compared with the CPU bake and the GPU time of the sharp sky. Run it after scene-modern.txt:
# ./tools/screenshots.ps1 -Targets neoforge-26.3,fabric-1.21.1 -Shots sky-galaxy.txt -OutDir build\sky-galaxy
wait 60
profile minimal
run time set 6000
wait 40
sky galaxy
skyturn 0
wait 160
# Yaw -45 looks south-east to the galaxy, pitch -46 looks up to its center.
fov 70
run tp @p 0 3000 0 -45 -46
wait 40
screenshot galaxy-01-day-fov70
fov 30
wait 20
screenshot galaxy-02-day-fov30
skycompare
wait 40
skytime
wait 40
run time set 18000
wait 40
fov 70
run tp @p ~ 3000 ~ -45 -25
wait 40
screenshot galaxy-03-night-low-fov70
run tp @p ~ 3000 ~ -45 -75
wait 40
screenshot galaxy-04-night-up-fov70
# The generators with no bloom, so that the shader draws all of them.
sky galaxy 4
skyturn 0
wait 160
fov 30
run tp @p ~ 3000 ~ -45 -46
wait 40
screenshot galaxy-05-nobloom-fov30
skycompare
wait 40
skytime
wait 40
skyrender fast
wait 160
screenshot galaxy-06-nobloom-fast-fov30
skyrender sharp
sky galaxy
skyturn 0
wait 160
# From the ground, where the terrain is in the view.
fov 70
run tp @p 0 ~ 0 -45 -30
run spreadplayers 0 0 0 1 false @p
wait 80
run tp @p ~ ~ ~ -45 -30
wait 40
screenshot galaxy-07-ground-fov70
quit
